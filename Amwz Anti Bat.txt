    if State.masterEnabled then
        statusLabel.Text = "STATUS: ONLINE"
        statusLabel.TextColor3 = GREEN

        statusLabel.Text = "STATUS: OFFLINE"

local function clearAntiDieConnections()
    if antiDieHealthConn then antiDieHealthConn:Disconnect() antiDieHealthConn = nil end
    if antiDieDiedConn then antiDieDiedConn:Disconnect() antiDieDiedConn = nil end
    if antiDieHeartbeatConn then antiDieHeartbeatConn:Disconnect() antiDieHeartbeatConn = nil end
end

local function bindResetButton(enabled)
    pcall(function()
        if enabled then
            StarterGui:SetCore("ResetButtonCallback", function() end)
        else
            StarterGui:SetCore("ResetButtonCallback", true)
        end
    end)
end

local function protectCharacter(character)
    clearAntiDieConnections()
    if not (State.antiDieEnabled and State.masterEnabled) then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    bindResetButton(true)
    humanoid.Health = humanoid.MaxHealth
    antiDieHealthConn = humanoid.HealthChanged:Connect(function(health)
        if State.antiDieEnabled and State.masterEnabled and health < humanoid.MaxHealth then
            humanoid.Health = humanoid.MaxHealth
        end
    end)
    antiDieDiedConn = humanoid.Died:Connect(function()
        if State.antiDieEnabled and State.masterEnabled then
            task.defer(function()
                if humanoid and humanoid.Parent then humanoid.Health = humanoid.MaxHealth end
            end)
        end
    end)
    antiDieHeartbeatConn = RunService.Heartbeat:Connect(function()
        if not (State.antiDieEnabled and State.masterEnabled) then return end
        if humanoid and humanoid.Parent then
            if humanoid.Health < humanoid.MaxHealth then humanoid.Health = humanoid.MaxHealth end
            if humanoid:GetState() == Enum.HumanoidStateType.Dead then
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
                humanoid.Health = humanoid.MaxHealth
            end
        end
    end)
end

local function setAntiDie(state)
    State.antiDieEnabled = state
    if state and State.masterEnabled then
        bindResetButton(true)
        if LocalPlayer.Character then protectCharacter(LocalPlayer.Character) end
    else
        clearAntiDieConnections()
        bindResetButton(false)
    end
end

local function startAntiBat()
    if antiBatConn then antiBatConn:Disconnect() end
    lastSafeCFrame, lastCheckTime = nil, 0
    antiBatConn = RunService.Heartbeat:Connect(function()
        if not (State.antiBatActive and State.masterEnabled) then return end
        local char = LocalPlayer.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then return end
        local range = State.rangeValue or 28
        local now = tick()
        if hrp.AssemblyLinearVelocity.Magnitude < 70 then
            lastSafeCFrame = hrp.CFrame
            lastCheckTime = now
        elseif hrp.AssemblyLinearVelocity.Magnitude > 110 and lastSafeCFrame and (now - lastCheckTime) < 1.5 then
            hrp.CFrame = lastSafeCFrame * CFrame.new(0, 0.1, 0)
        end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local eHrp = plr.Character:FindFirstChild("HumanoidRootPart")
                local tool = plr.Character:FindFirstChildWhichIsA("Tool")
                if eHrp and tool and tool.Name:lower():find("bat") then
                    if (hrp.Position - eHrp.Position).Magnitude < range then
                        if State.evasionMode == "ORBITAL" then
                            local angle = math.rad(tick() * 500)
                            hrp.CFrame = hrp.CFrame * CFrame.new(math.sin(angle) * 3, 0, math.cos(angle) * 3)
                        else
                            local offset = (hrp.Position - eHrp.Position).Unit * (range + 5)
                            hrp.CFrame = CFrame.new(eHrp.Position + offset + Vector3.new(0, 2, 0))
                        end
                    end
                end
            end
        end
    end)
end

local function stopAntiBat()
    if antiBatConn then antiBatConn:Disconnect() antiBatConn = nil end
end

local function startFreeze()
    if freezeConn then freezeConn:Disconnect() end
    freezeConn = RunService.Heartbeat:Connect(function()
        if not (State.freezeOthers and State.masterEnabled) then return end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                if hrp then hrp.AssemblyLinearVelocity = Vector3.zero hrp.AssemblyAngularVelocity = Vector3.zero end
            end
        end
    end)
end

local function stopFreeze()
    if freezeConn then freezeConn:Disconnect() freezeConn = nil end
end

local function startFling()
    if flingConn then flingConn:Disconnect() end
    flingConn = RunService.Heartbeat:Connect(function()
        if not (State.flingEngine and State.masterEnabled) then return end
        local myHrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if not myHrp then return end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local eHrp = plr.Character:FindFirstChild("HumanoidRootPart")
                if eHrp and (myHrp.Position - eHrp.Position).Magnitude < (State.rangeValue or 28) then
                    eHrp.AssemblyLinearVelocity = (eHrp.Position - myHrp.Position).Unit * 150 + Vector3.new(0, 80, 0)
                end
            end
        end
    end)
end

local function stopFling()
    if flingConn then flingConn:Disconnect() flingConn = nil end
end

local dragInfo = {active = false}
mainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragInfo.active = true
        dragInfo.start = input.Position
        dragInfo.startPos = mainFrame.Position
    end
end)
mainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInfo.input = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInfo.input and dragInfo.active then
        local delta = input.Position - dragInfo.start
        mainFrame.Position = UDim2.new(dragInfo.startPos.X.Scale, dragInfo.startPos.X.Offset + delta.X, dragInfo.startPos.Y.Scale, dragInfo.startPos.Y.Offset + delta.Y)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragInfo.active = false
    end
end)
